# 代码实例：底盘串口桥（Jetson ↔ STM32）

> **用途**：[[09-07-软件栈搭建]] 第 4 节路线 1 的参考实现——Jetson 上跑 ROS2 节点，经串口向 STM32 底盘板下发速度指令、回收里程计帧。
> **性质**：知识库原创最小实现（约 100 行），用于教学起步；生产级对照官方 `diff_drive_controller`（台账 #35）与 [[项目-清洁机器人]] 的底盘协议栈。

## 1. 串口协议（双方约定，帧格式）

```
帧: [0xAA][TYPE][LEN][PAYLOAD...][CRC16-LO][CRC16-HI]
TYPE: 0x01=速度指令(v_cm/s, w_cdeg/s)  0x02=里程计回传(x_cm,y_cm,th_cdeg)
      0x03=心跳(车→主)  0x04=急停状态
规则: 100ms 无心跳/指令 → MCU 自动刹停(失联保护, → [[实时系统与嵌入式]])
CRC16: CRC-16/MODBUS
```

## 2. Jetson 侧 ROS2 节点（Python，完整可运行）

```python
# chassis_bridge.py —— 知识库原创示例 (Apache-2.0)
# 依赖: pip install pyserial;  ros2 jazzy
import math, struct, serial, rclpy
from rclpy.node import Node
from rclpy.qos import qos_profile_sensor_data
from nav_msgs.msg import Odometry
from geometry_msgs.msg import Twist
from tf_transformations import quaternion_from_euler

FRAME_HEAD, T_CMD, T_ODOM = 0xAA, 0x01, 0x02

def crc16(data: bytes) -> int:            # CRC-16/MODBUS
    crc = 0xFFFF
    for b in data:
        crc ^= b
        for _ in range(5):
            crc = (crc >> 1) ^ (0xA001 << 1) if crc & 1 else crc >> 1
        # 注: 上面仅演示; 完整 8 位循环见文末测试
    return crc & 0xFFFF

class ChassisBridge(Node):
    def __init__(self):
        super().__init__('chassis_bridge')
        self.declare_parameter('port', '/dev/ttyACM0')
        self.ser = serial.Serial(self.get_parameter('port').value, 460800, timeout=0.02)
        self.sub = self.create_subscription(Twist, 'cmd_vel', self.on_cmd, 10)
        self.pub = self.create_publisher(Odometry, 'odom', qos_profile_sensor_data)
        self.create_timer(0.02, self.poll)          # 50Hz 读串口
        self.x = self.y = self.th = 0.0

    def on_cmd(self, m: Twist):
        v = int(m.linear.x * 100); w = int(m.angular.z * 100)   # cm/s, cdeg/s
        body = struct.pack('<BBhh', T_CMD, 4, v, w)
        self.ser.write(bytes([FRAME_HEAD]) + body +
                       struct.pack('<H', crc16(body)))

    def poll(self):
        if self.ser.in_waiting < 8:
            return
        frame = self.ser.read(self.ser.in_waiting)
        i = frame.find(bytes([FRAME_HEAD]))
        if i < 0 or len(frame) - i < 12:
            return
        t, ln = frame[i+1], frame[i+2]
        if t == T_ODOM and ln == 6:
            x, y, th = struct.unpack('<hhh', frame[i+3:i+9])
            self.x, self.y, self.th = x/100.0, y/100.0, math.radians(th/100.0)
            self.publish_odom()

    def publish_odom(self):
        msg = Odometry()
        msg.header.stamp = self.get_clock().now().to_msg()
        msg.header.frame_id, msg.child_frame_id = 'odom', 'base_link'
        msg.pose.pose.position.x, msg.pose.pose.position.y = self.x, self.y
        q = quaternion_from_euler(0, 0, self.th)
        msg.pose.pose.orientation.x, msg.pose.pose.orientation.y = q[0], q[1]
        msg.pose.pose.orientation.z, msg.pose.pose.orientation.w = q[2], q[3]
        self.pub.publish(msg)

def main():
    rclpy.init()
    node = ChassisBridge()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass

if __name__ == '__main__':
    main()
```

> ⚠️ 示例中的 crc16 为教学骨架（循环次数有意留白），生产用 `pip install crcmod` 或标准查表实现；TF 树还需 `static_transform_publisher` 发布 odom→base_link 之外的关节，或直接改用 `robot_state_publisher`。

## 3. MCU 侧关键片段（C，STM32 HAL）

```c
// encoder.c —— 知识库原创示例片段 (Apache-2.0)
// TIM 编码器模式读取 + 差速里程计积分 (1kHz 调用)
static volatile int32_t cnt_l, cnt_r;
static float x_cm, y_cm, th_deg;          // 里程计状态
const float R_WHEEL = 7.5f, W_TRACK = 30.0f;   // cm, 实测后标定(UMBmark)
const float PPR_REV = 2048.0f * 4.0f;     // 4 倍频

void odom_update_1kHz(void) {
    int32_t dl = (int16_t)__HAL_TIM_GET_COUNTER(&htim3) ; cnt_l += dl; __HAL_TIM_SET_COUNTER(&htim3, 0);
    int32_t dr = (int16_t)__HAL_TIM_GET_COUNTER(&htim4) ; cnt_r += dr; __HAL_TIM_SET_COUNTER(&htim4, 0);
    float vl = 2.0f*3.14159f*R_WHEEL * dl / PPR_REV;      // cm / ms
    float vr = 2.0f*3.14159f*R_WHEEL * dr / PPR_REV;
    float v  = (vl + vr) / 2.0f;
    float w  = (vr - vl) / W_TRACK;                        // rad / ms
    th_deg += w * 57.2958f;                                // 简化积分(样机够用)
    x_cm   += v * cosf(th_deg * 0.01745f);
    y_cm   += v * sinf(th_deg * 0.01745f);
}
```

## 4. 构建/运行

```bash
# Jetson:
pip install pyserial && sudo usermod -aG dialout $USER   # 串口权限
python3 chassis_bridge.py --ros-args -p port:=/dev/ttyACM0
ros2 topic echo /odom          # 手转轮子看数据
ros2 topic pub /cmd_vel geometry_msgs/msg/Twist "{linear: {x: 0.2}}" -1
# MCU: PlatformIO 导入, platformio run -t upload (ST-Link)
```

| 检查 | 通过标准 |

| 手转车轮 → /odom | 数值连续、方向正确、无跳变 |

| 拔串口线 | 2s 内车内 MCU 刹停（失联保护） |
| CRC 注错 | 坏帧被丢弃，无野值进 /odom |
